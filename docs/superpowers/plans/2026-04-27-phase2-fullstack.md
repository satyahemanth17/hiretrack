# HireTrack Phase 2 — Full Stack Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Complete HireTrack from a backend-only API into a fully deployable full-stack application with Alembic migrations, BFF, Next.js frontend, Docker Compose, CI/CD, and README.

**Architecture:** A Node.js/Express TypeScript BFF on port 4000 proxies all frontend traffic to FastAPI on port 8000, enforcing rate limits and CORS. The Next.js 14 frontend on port 3000 communicates exclusively with the BFF. PostgreSQL 16 is the database, orchestrated via Docker Compose.

**Tech Stack:** Python 3.14 + FastAPI + Alembic (backend), Node 20 + Express + TypeScript (BFF), Next.js 14 + TypeScript + Tailwind CSS + @dnd-kit + recharts (frontend), PostgreSQL 16, Docker Compose, GitHub Actions.

Create `/Users/satya/Projects/hiretrack/frontend/lib/api.ts`:

```typescript
const BFF_URL = process.env.NEXT_PUBLIC_BFF_URL ?? "http://localhost:4000";

export function getToken(): string | null {
  if (typeof window === "undefined") return null;
  return localStorage.getItem("token");
}

export function setToken(token: string): void {
  localStorage.setItem("token", token);
}

export function clearToken(): void {
  localStorage.removeItem("token");
}

async function apiFetch<T>(path: string, options: RequestInit = {}): Promise<T> {
  const token = getToken();
  const { headers: callerHeaders, ...rest } = options;
  const headers: Record<string, string> = {
    "Content-Type": "application/json",
    ...((callerHeaders as Record<string, string>) ?? {}),
  };
  if (token) headers.Authorization = `Bearer ${token}`;

  const res = await fetch(`${BFF_URL}${path}`, { ...rest, headers });
  if (!res.ok) {
    const text = await res.text();
    throw new Error(`API ${res.status}: ${text}`);
  }
  if (res.status === 204) return undefined as T;
  return res.json() as Promise<T>;
}

export type ApplicationStatus =
  | "applied"
  | "phone_screen"
  | "interview"
  | "offer"
  | "rejected"
  | "withdrawn";

export interface Application {
  id: string;
  company: string;
  role: string;
  status: ApplicationStatus;
  applied_date: string;
  location: string | null;
  job_url: string | null;
  salary_min: number | null;
  salary_max: number | null;
  follow_up_date: string | null;
  notes: string | null;
  job_description: string | null;
  user_id: number;
  created_at: string;
  updated_at: string;
}

export interface ApplicationCreate {
  company: string;
  role: string;
  status: ApplicationStatus;
  applied_date: string;
  location?: string;
  job_url?: string;
  salary_min?: number;
  salary_max?: number;
  follow_up_date?: string;
  notes?: string;
  job_description?: string;
}

export interface FunnelItem {
  status: string;
  count: number;
}

export interface TimelineItem {
  date: string;
  count: number;
}

export interface MatcherResult {
  score: number;
  matched: string[];
  missing: string[];
}

export function listApplications(limit = 100): Promise<Application[]> {
  return apiFetch<Application[]>(`/api/applications?limit=${limit}`);
}

export function createApplication(data: ApplicationCreate): Promise<Application> {
  return apiFetch<Application>("/api/applications", {
    method: "POST",
    body: JSON.stringify(data),
  });
}

export function updateApplication(
  id: string,
  data: Partial<ApplicationCreate>
): Promise<Application> {
  return apiFetch<Application>(`/api/applications/${id}`, {
    method: "PATCH",
    body: JSON.stringify(data),
  });
}

export function deleteApplication(id: string): Promise<void> {
  return apiFetch<void>(`/api/applications/${id}`, { method: "DELETE" });
}

export function getFunnel(): Promise<FunnelItem[]> {
  return apiFetch<FunnelItem[]>("/api/analytics/funnel");
}

export function getTimeline(): Promise<TimelineItem[]> {
  return apiFetch<TimelineItem[]>("/api/analytics/timeline");
}

export function analyzeKeywords(
  resume: string,
  job_description: string
): Promise<MatcherResult> {
  return apiFetch<MatcherResult>("/api/matcher/analyze", {
    method: "POST",
    body: JSON.stringify({ resume, job_description }),
  });
}
```

Create `/Users/satya/Projects/hiretrack/.github/workflows/ci.yml`:

```yaml
name: CI

on:
  push:
    branches: [main, dev]
  pull_request:
    branches: [main]

jobs:
  test-backend:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.14"
          allow-prereleases: true
      - name: Install dependencies
        run: pip install -r backend/requirements.txt
      - name: Run tests
        run: cd backend && pytest -v

  test-bff:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "20"
          cache: "npm"
          cache-dependency-path: bff/package-lock.json
      - name: Install dependencies
        run: cd bff && npm ci
      - name: TypeScript check
        run: cd bff && npx tsc --noEmit

  test-frontend:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "20"
          cache: "npm"
          cache-dependency-path: frontend/package-lock.json
      - name: Install dependencies
        run: cd frontend && npm ci
      - name: TypeScript check
        run: cd frontend && npx tsc --noEmit
      - name: Build
        run: cd frontend && npm run build

  lint-docker:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Validate Docker Compose config
        run: docker compose config
```

Browser → Next.js (3000) → BFF (4000) → FastAPI (8000) → PostgreSQL (5432)

                                      └─ CORS (localhost:3000)

| Layer | Technology |
|-------|-----------|
| Frontend | Next.js 14, TypeScript, Tailwind CSS, @dnd-kit, recharts |
| BFF | Node.js 20, Express 4, TypeScript, express-rate-limit |
| Backend | FastAPI, Python 3.14, SQLAlchemy 2, Pydantic v2, Alembic |
| Database | PostgreSQL 16 |
| Auth | GitHub OAuth 2.0 + JWT (HS256, 60-min expiry) |
| Container | Docker + Docker Compose |
| CI/CD | GitHub Actions |
